    <!-- Dynamic Loader Script -->
    <script>
        let KANBAN_STORAGE_KEY = 'default_kanban_tasks';
        let currentProjectData = null;

        document.addEventListener('DOMContentLoaded', async () => {
            const urlParams = new URLSearchParams(window.location.search);
            const projectId = urlParams.get('id');

            if (!projectId) {
                alert('No project specified!');
                window.location.href = 'projects.html';
                return;
            }

            // Set unique storage keys for this specific project's board & scratchpad
            KANBAN_STORAGE_KEY = `project_kanban_${projectId}`;
            const kanbanSection = document.getElementById('project-kanban-section');
            if (kanbanSection) kanbanSection.setAttribute('data-storage-key', KANBAN_STORAGE_KEY);

            // Fetch project details from Supabase
            const { data: project, error } = await supabaseClient
                .from('projects')
                .select('*')
                .eq('id', projectId)
                .single();

            if (error || !project) {
                console.error('Error fetching project:', error);
                document.getElementById('project-header-title').textContent = 'Project Not Found';
                return;
            }

            // Store project data locally for editing
            currentProjectData = project;

            // Populate the page with project data
            document.getElementById('browser-title').textContent = `${project.title} - My Control Center`;
            document.getElementById('project-header-title').textContent = project.title;
            document.getElementById('project-card-title').textContent = project.title;
            document.getElementById('project-description').textContent = project.description || 'No description provided.';

            // Handle optional external link banner
            if (project.external_link) {
                const banner = document.getElementById('external-banner-wrapper');
                const linkEl = document.getElementById('external-banner-link');
                linkEl.href = project.external_link;
                banner.style.display = 'flex';
            }
        });

        function openEditProjectModal() {
            if (!currentProjectData) return;
            document.getElementById('edit-title').value = currentProjectData.title || '';
            document.getElementById('edit-description').value = currentProjectData.description || '';
            document.getElementById('edit-external-link').value = currentProjectData.external_link || '';
            document.getElementById('edit-error-msg').textContent = '';
            document.getElementById('edit-project-modal').style.display = 'flex';
        }

        function closeEditProjectModal() {
            document.getElementById('edit-project-modal').style.display = 'none';
        }

        async function handleSaveProject(event) {
            event.preventDefault();
            const urlParams = new URLSearchParams(window.location.search);
            const projectId = urlParams.get('id');

            const updatedTitle = document.getElementById('edit-title').value.trim();
            const updatedDescription = document.getElementById('edit-description').value.trim();
            const updatedExternalLink = document.getElementById('edit-external-link').value.trim();

            const { error } = await supabaseClient
                .from('projects')
                .update({
                    title: updatedTitle,
                    description: updatedDescription,
                    external_link: updatedExternalLink || null
                })
                .eq('id', projectId);

            if (error) {
                console.error('Error updating project:', error);
                document.getElementById('edit-error-msg').textContent = 'Failed to update project: ' + error.message;
                return;
            }

            // Update local state and DOM elements instantly
            currentProjectData.title = updatedTitle;
            currentProjectData.description = updatedDescription;
            currentProjectData.external_link = updatedExternalLink;

            document.getElementById('browser-title').textContent = `${updatedTitle} - My Control Center`;
            document.getElementById('project-header-title').textContent = updatedTitle;
            document.getElementById('project-card-title').textContent = updatedTitle;
            document.getElementById('project-description').textContent = updatedDescription || 'No description provided.';

            const banner = document.getElementById('external-banner-wrapper');
            const linkEl = document.getElementById('external-banner-link');
            if (updatedExternalLink) {
                linkEl.href = updatedExternalLink;
                banner.style.display = 'flex';
            } else {
                banner.style.display = 'none';
            }

            closeEditProjectModal();
        }
    </script>
